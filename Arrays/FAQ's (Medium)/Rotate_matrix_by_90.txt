class Solution {
public:
    void rotateMatrix(vector<vector<int>>& matrix) {
        //tranpose and then reverse the rows - trick
        for(int i=0; i<matrix.size(); i++){
            for(int j=i+1; j<matrix.size(); j++){
                swap(matrix[i][j] , matrix[j][i]);
            }
        }
        for(int i=0; i<matrix.size(); i++){
            reverse(matrix[i].begin() , matrix[i].end());
        }
    }
};
