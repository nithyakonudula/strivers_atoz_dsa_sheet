class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        vector<pair<int, int>> arr;
        for(int i=0; i<nums.size(); i++){
            arr.push_back({nums[i], i});
        }

        sort(arr.begin(), arr.end());

        int start = 0;
        int last = arr.size()-1;

        while(start < last){
            int sum = arr[start].first + arr[last].first ;
            if(sum > target)
                last--;
            else if(sum < target)
                start++;
            else 
                return {arr[start].second , arr[last].second};
        }
        return {};
    }
};
